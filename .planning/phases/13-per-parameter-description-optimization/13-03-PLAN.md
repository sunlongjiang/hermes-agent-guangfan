  ```python
  # ── Joint Tool-Param Metric (Phase 13: D-10, D-17) ───────────────────────────

  import json  # move to top if not already imported; check existing imports first


  # Normalization rule sourced from Wave 0 dataset inspection
  # (see .planning/phases/13-.../13-correct-params-type-inspection.txt).
  # Default policy: "strip_only" — applies str.strip() on both sides when both
  # values are strings; identity comparison otherwise. If inspection recommended
  # "strip_plus_coerce", uncomment the numeric-coerce block marked below.
  _NORMALIZATION_RULE = "strip_only"  # SET ACCORDING TO 13-01 OUTPUT


  def _normalize_param_value(v):
      """Light normalization for joint-metric param value comparison.

      Policy (evidence-based from 13-01 Wave 0 inspection):
      - Strip leading/trailing whitespace for strings.
      - Numeric coercion ONLY if NORMALIZATION_RULE == 'strip_plus_coerce'.
      - Preserve None / bool / empty-dict identity.
      """
      if isinstance(v, str):
          return v.strip()
      if _NORMALIZATION_RULE == "strip_plus_coerce":
          # Enable via 13-01 evidence only — commented out by default to preserve
          # strict exact-match semantics.
          if isinstance(v, bool):
              return v  # keep bool distinct from int per PEP 285
          if isinstance(v, (int, float)):
              return v
      return v


  def _parse_selected_params_json(raw) -> Optional[dict]:
      """Parse a selected_params value into a dict, or None on malformed input.

      Returns:
          dict if the raw string parses to a JSON object (or the input is already
          a dict); None on any parse failure / non-object / non-dict result.
      """
      if raw is None:
          return None
      if isinstance(raw, dict):
          return raw
      if not isinstance(raw, str):
          try:
              return dict(raw)
          except (TypeError, ValueError):
              return None
      try:
          parsed = json.loads(raw)
      except (json.JSONDecodeError, TypeError):
          return None
      if not isinstance(parsed, dict):
          return None
      return parsed


  def _param_match_score(
      predicted_params: Optional[dict], correct_params: dict
  ) -> float:
      """Return 1.0 iff dict-equal after light normalization; 0.0 otherwise.

      Pitfall 4 mitigation: strip-only normalization applied to both sides so
      trailing whitespace in LLM output does not spuriously break exact match.
      """
      if predicted_params is None:
          return 0.0
      # Same set of keys, same normalized values.
      if set(predicted_params.keys()) != set(correct_params.keys()):
          return 0.0
      for k, v_correct in correct_params.items():
          v_pred = predicted_params.get(k)
          if _normalize_param_value(v_correct) != _normalize_param_value(v_pred):
              return 0.0
      return 1.0


  def joint_tool_param_metric(
      example: dspy.Example,
      prediction: dspy.Prediction,
      trace=None,
      pred_name=None,
      pred_trace=None,
  ) -> float:
      """Joint tool + param metric (D-10, D-17).

      Returns:
          float in [0.0, 1.0]: 0.5 * tool_match + 0.5 * param_match, both
          exact-match after strip/lower normalization (tool_match) or
          strip-only value comparison (param_match).

      Contract (D-10):
          - tool_match = 1.0 iff selected_tool.strip().lower() == correct_tool.strip().lower()
          - param_match = 1.0 iff parsed selected_params dict-equals correct_params
          - Malformed JSON in selected_params → param_match = 0.0 (Pitfall 7)
          - Signature is GEPA's 5-param form (gepa.py:368 binding check).
      """
      selected_tool = (getattr(prediction, "selected_tool", "") or "").strip().lower()
      correct_tool = (getattr(example, "correct_tool", "") or "").strip().lower()
      tool_match = 1.0 if selected_tool == correct_tool else 0.0

      correct_params = getattr(example, "correct_params", {}) or {}
      raw = getattr(prediction, "selected_params", "")
      predicted = _parse_selected_params_json(raw)
      param_match = _param_match_score(predicted, correct_params)

      return 0.5 * tool_match + 0.5 * param_match


  def joint_tool_param_metric_with_feedback(
      example: dspy.Example,
      prediction: dspy.Prediction,
      trace=None,
      pred_name=None,
      pred_trace=None,
  ) -> dspy.Prediction:
      """GEPA-facing metric returning ScoreWithFeedback (Pitfall 7 mitigation).

      Returns:
          dspy.Prediction with:
              - score (float): identical to joint_tool_param_metric()'s return.
              - feedback (str): short human-readable reason string that GEPA's
                reflection_lm uses when proposing new param descriptions.

      Why dspy.Prediction and not a bare dict:
          DSPy 3.1.3's GEPA reflection code paths use attribute access
          (`result.score`, `result.feedback`) on the metric return. A bare
          `{"score": ..., "feedback": ...}` dict would require subscript
          access and is NOT guaranteed compatible across all GEPA 3.1.3
          internals. `dspy.Prediction` exposes both attribute AND dict-style
          access, so it is the safe cross-path return type.

      Important: do NOT use this for acceptance / holdout scoring — use
      joint_tool_param_metric (plain float) instead. Mixing the two in
      cross-tool regression gates would break the CrossToolRegressionChecker
      contract (expects float input).
      """
      score = joint_tool_param_metric(
          example, prediction, trace, pred_name, pred_trace
      )
      correct_tool = (getattr(example, "correct_tool", "") or "").strip().lower()
      selected_tool = (getattr(prediction, "selected_tool", "") or "").strip().lower()
      correct_params = getattr(example, "correct_params", {}) or {}
      raw = getattr(prediction, "selected_params", "")
      predicted = _parse_selected_params_json(raw)

      fb_parts: list[str] = []
      if selected_tool != correct_tool:
          fb_parts.append(
              f"Wrong tool: picked '{selected_tool}', expected '{correct_tool}'."
          )
      if predicted is None:
          fb_parts.append(
              "selected_params was not valid JSON; return a JSON object like "
              "'{\"pattern\":\"foo\"}'."
          )
      elif predicted != correct_params:
          missing = sorted(set(correct_params) - set(predicted))
          extra = sorted(set(predicted) - set(correct_params))
          bad_val = sorted(
              k for k in set(correct_params) & set(predicted)
              if _normalize_param_value(correct_params[k])
                 != _normalize_param_value(predicted[k])
          )
          fb_parts.append(
              f"Param mismatch -- missing keys: {missing}; extra keys: {extra};"
              f" wrong values for keys: {bad_val}."
          )
      if not fb_parts:
          fb_parts.append("Perfect match.")

      return dspy.Prediction(score=float(score), feedback=" ".join(fb_parts))
  ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| LLM-generated selected_params JSON → json.loads | Untrusted LLM text parsed; any exception must NOT crash GEPA loop |
| metric output → GEPA reflection_lm | feedback string fed back to LLM; should not leak training-data verbatim |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-13-08 | Denial of Service | json.loads on unbounded LLM output | mitigate | Wrap json.loads in try/except (json.JSONDecodeError, TypeError); malformed → None → score 0.0 (no crash, no retry loop). Output size bounded by LLM max_tokens (configured in EvolutionConfig) — no parser pathology on normal outputs. |
| T-13-09 | Information Disclosure | feedback string includes correct_params keys/values | mitigate | Feedback only names keys and summarizes diff class ("missing keys", "wrong values"); does NOT echo the full correct_params dict verbatim. In a single-user CLI this is acceptable; Phase 22 continuous loop will need tighter redaction. Accepted risk for Phase 13. |
| T-13-10 | Tampering | _NORMALIZATION_RULE constant | accept | Hardcoded string chosen by planner based on 13-01 evidence; no user-facing override. Changing the rule is a code review event, not runtime config. |
