    **Helper 1: _normalize_task_hash** (D-13):

    ```python
    def _normalize_task_hash(task: str) -> str:
        """Return sha256(strip + lower + collapse_whitespace(task))[:16].

        Used to dedup the same task across signal sources and to derive a
        deterministic train/val/holdout bucket. The 16-char hex slice is per
        CONTEXT D-13 specifics — not a security boundary.
        """
        norm = re.sub(r"\s+", " ", (task or "").lower()).strip()
        return hashlib.sha256(norm.encode("utf-8")).hexdigest()[:16]
    ```

    **Helper 2: _hash_to_split** (D-13):
    ```python
    def _hash_to_split(h: str) -> str:
        """Bucket per CONTEXT D-13: <70 train / <85 val / else holdout."""
        bucket = int(h[:8], 16) % 100
        if bucket < 70:
            return "train"
        if bucket < 85:
            return "val"
        return "holdout"
    ```

    **Helper 3: _multiplier_for** (D-11):
    ```python
    def _multiplier_for(signals: list[str], override: Optional[dict[str, int]] = None) -> int:
        """Return max multiplier across hit signals; default 1 if no signals match.

        Per D-11 max policy (multi-source hits do NOT accumulate). `override`
        merges over DEFAULT_MULTIPLIER on per-key basis.
        """
        merged = dict(DEFAULT_MULTIPLIER)
        if override:
            merged.update({k: v for k, v in override.items() if k in DEFAULT_MULTIPLIER})
        hits = [merged[s] for s in signals if s in merged]
        return max(hits) if hits else 1
    ```

    ```python
    def _load_jsonl_skip_bad(path: Path) -> tuple[list[dict], int]:
        """Read JSONL line-by-line; return (rows, skipped_count).

        Per D-18 minimal subset: replicates the per-line try/except pattern
        already in external_importers.py:185-188 (ClaudeCodeImporter). Warns
        via Rich console when skipped/total > 5%.

        EXPLICITLY NOT in scope: EvalDataset.load / GoldenDatasetLoader.load /
        ToolSelectionDataset.load — those are owned by v2-STAB-01.

        Args:
            path: Path to a .jsonl file. Missing file returns ([], 0).

        Returns:
            (rows, skipped_count) tuple.
        """
        rows: list[dict] = []
        skipped = 0
        if not path.exists():
            return rows, 0
        with open(path) as f:
            for line in f:
                if not line.strip():
                    continue
                try:
                    rows.append(json.loads(line))
                except json.JSONDecodeError:
                    skipped += 1
        total = len(rows) + skipped
        if total and skipped / total > JSONL_BAD_LINE_WARN_THRESHOLD:
            console.print(
                f"[yellow]⚠ {path.name}: skipped {skipped}/{total} bad JSONL "
                f"lines ({skipped / total * 100:.1f}%)[/yellow]"
            )
        return rows, skipped
    ```

    ```python
              2) per session, call _extract_error_retry / _extract_user_correction /
                 _extract_oracle_disagreement (per --signals subset)

                 not in current_tools name set

                 downstream_context)

              7) bucket split (D-13)

              9) return list[ToolSelectionExample]

        Args:
            config: EvolutionConfig providing judge_model.
            signals: Subset of {"error_retry","user_correction","oracle_disagreement"}.

            multiplier_override: Per-signal multiplier dict (D-12 CLI flag pass-through).
            baseline_module: Optional ToolModule for C signal oracle (D-04). None = skip C.

        class ConfirmMisselection(dspy.Signature):
            """Decide whether the originally used tool was a misselection.

            Given the user's task, the available tools, and the downstream context
            after the tool call (the next 1-3 turns), decide:
              - verdict: 'confirm_misselection' if a different available tool would
                clearly have been more appropriate; else 'false_positive'.
              - correct_tool: when verdict='confirm_misselection', the name of the
                better tool (must appear in available_tools_summary). Otherwise
                may be empty or echo originally_used_tool.
              - rationale: one sentence justification.

            When uncertain, prefer 'false_positive' (drop the candidate).
            """
            task_description: str = dspy.InputField(
                desc="The user task that triggered the assistant tool call",
            )
            available_tools_summary: str = dspy.InputField(
                desc="Newline-separated list of '- name: description' for all current tools",
            )
            originally_used_tool: str = dspy.InputField(
                desc="Name of the tool the assistant actually called",
            )
            signal_source: str = dspy.InputField(
                desc="Which heuristic flagged this candidate: error_retry|user_correction|oracle_disagreement",
            )
            downstream_context: str = dspy.InputField(
                desc="Summary of the next 1-3 messages after the tool call",
            )
            verdict: str = dspy.OutputField(
                desc="'confirm_misselection' or 'false_positive'; default 'false_positive' when unsure",
            )
            correct_tool: str = dspy.OutputField(
                desc="When confirm_misselection: name of the better tool from available_tools_summary",
            )
            rationale: str = dspy.OutputField(
                desc="One-sentence justification for the verdict",
            )

        class DetectUserCorrection(dspy.Signature):
            """LLM 二判 for A signal: decide whether a user message is genuinely correcting tool choice.

            Keyword regex pre-filter has already matched. This judge resolves
            false-positives where the keyword appears in unrelated context.
            """
            user_message: str = dspy.InputField(desc="The user message after the tool_call")
            preceding_tool_call: str = dspy.InputField(desc="Name + args of the tool call being potentially corrected")
            is_correction: bool = dspy.OutputField(desc="True if user is correcting the tool choice")

        # User-correction keyword seeds (CONTEXT specifics + Open Questions resolution)
        _USER_CORRECTION_PATTERNS: list[str] = [
            r"不对",
            r"错了",
            r"不应该",
            r"应该用",
            r"应该是",
            r"换一个",
            r"换工具",
            r"不是要",
            r"\bwrong tool\b",
            r"\bshould have used\b",
            r"\buse \w+ instead\b",
            r"\bthat'?s not right\b",
        ]

        def __init__(

            signals: Optional[list[str]] = None,
            multiplier_override: Optional[dict[str, int]] = None,
            baseline_module=None,  # ToolModule | None

            self.signals = signals or list(VALID_SIGNALS)
            self.multiplier_override = multiplier_override or {}
            self.baseline_module = baseline_module
            self.judge = dspy.ChainOfThought(self.ConfirmMisselection)

            # Audit counters (filled by mine())
            self.metrics: dict = self._fresh_metrics()
    ```

    ```python
        # ── B signal: error_retry ─────────────────────────────────────────
        def _is_tool_error(self, parsed_content, raw: str) -> bool:
            """B signal helper: tolerate three forms of tool error (Pitfall 2)."""
            if isinstance(parsed_content, dict):
                ec = parsed_content.get("exit_code")
                if isinstance(ec, int) and ec != 0:
                    return True
                err = parsed_content.get("error")
                if isinstance(err, str) and err.strip() and err.strip().lower() != "null":
                    return True
                return False
            return any(kw in raw for kw in ("Traceback", "Exception:", "Error:", '"error":'))

        def _extract_error_retry(self, messages: list[dict], session_path: str,
                                 current_tool_names: set[str]) -> list[Candidate]:
            """Extract B signal candidates: assistant tool_call → tool error → next assistant tool_call (different name) → success.

            Chunk boundary = next user message (Pitfall 3). Take only the FIRST
            error→success switch per chunk; don't recurse.
            """
            cands: list[Candidate] = []
            current_user_task = ""
            i = 0
            while i < len(messages):
                m = messages[i]
                role = m.get("role")
                if role == "user":
                    current_user_task = (m.get("content") or "").strip()
                    i += 1
                    continue
                if role == "assistant":
                    tool_calls = m.get("tool_calls")
                    if not isinstance(tool_calls, list) or not tool_calls:
                        i += 1
                        continue
                    # Look for: this assistant has tool_calls, next message is tool with error,
                    # then somewhere within the chunk another assistant tool_call (different name) succeeds.
                    for tc in tool_calls:
                        try:
                            tc_id = tc.get("id") or ""
                            fn = tc.get("function") or {}
                            tool_name = fn.get("name")
                            if not tool_name or tc_id == "":
                                continue
                            # find matching tool message
                            tool_msg = self._find_tool_message_by_id(messages, i + 1, tc_id)
                            if tool_msg is None:
                                continue
                            parsed_content = self._parse_tool_content(tool_msg.get("content", ""))
                            if not self._is_tool_error(parsed_content, tool_msg.get("content", "")):
                                continue
                            # search for next assistant tool_call (different name + success) until next user
                            recovery = self._find_recovery_tool_call(
                                messages, i + 1, tool_name, current_tool_names
                            )
                            if recovery is None:
                                continue
                            cands.append(Candidate(
                                task=current_user_task,
                                session_path=session_path,
                                originally_used_tool=tool_name,
                                available_tools=sorted(current_tool_names),
                                tool_call_id=tc_id,
                                signal="error_retry",
                                downstream_context=self._summarize_context(
                                    messages, i, recovery_idx=recovery
                                ),
                            ))
                            break  # only first error→success switch per assistant turn
                        except Exception:
                            continue
                i += 1
            return cands
    ```

    给出 `_find_tool_message_by_id`, `_parse_tool_content` (json.loads + try/except), `_find_recovery_tool_call`, `_summarize_context` 四个私有 helper（每个 5–15 行）。所有读取使用 `.get()` / `isinstance` 检查避免 KeyError 与 TypeError —— 见 Pitfall 1 schema-tolerant 要求。

    **A signal extractor** 类似 B，关键词 regex (`_USER_CORRECTION_PATTERNS`) 命中后送 `DetectUserCorrection` LLM 二判 (`_parse_bool(getattr(result, "is_correction", None))`)。LLM 调用前后用同一 try/except 模式（PATTERNS §Pattern C）：异常 → 当作未纠正，跳过 candidate。Pitfall 7 嘱：关键词预过滤要尽量收紧；Open Question 2 嘱：取最后一个 tool_call。

    **C signal extractor**：仅当 `self.baseline_module is not None` 时启用。每个 successful tool_call → 调 `self.baseline_module(task_description=task)` → 取 `.selected_tool` → 若 ≠ session 实际 `function.name` 即生成 candidate。同样包 try/except → 异常 skip candidate。

    ```python
        def _judge_candidate(self, cand: Candidate) -> Verdict:
            """LLM judge per candidate. Fail-closed → false_positive on any error."""
            self.metrics["judge_calls"] += 1
            self.metrics["judge_calls_by_signal"][cand.signal] += 1
            lm = dspy.LM(self.config.judge_model, **self.config.get_lm_kwargs())
            tools_summary = "\n".join(f"- {n}" for n in cand.available_tools)
            try:
                with dspy.context(lm=lm):
                    result = self.judge(
                        task_description=cand.task,
                        available_tools_summary=tools_summary,
                        originally_used_tool=cand.originally_used_tool,
                        signal_source=cand.signal,
                        downstream_context=cand.downstream_context,
                    )
            except Exception as e:
                return Verdict(label="false_positive", correct_tool="",
                               rationale=f"judge_error: {e}")
            label = (str(getattr(result, "verdict", "")) or "").strip().lower()
            if label not in ("confirm_misselection", "false_positive"):
                label = "false_positive"  # CONCERNS M4 fail-closed
            correct = str(getattr(result, "correct_tool", "") or "").strip()
            rationale = str(getattr(result, "rationale", "") or "").strip()
            # Sanity: when confirmed, correct_tool must be a known tool name.
            if label == "confirm_misselection" and correct not in cand.available_tools:
                label = "false_positive"  # gate also on schema mismatch
            return Verdict(label=label, correct_tool=correct, rationale=rationale)
    ```

    ```python
        def mine(self, sessions_dir: Path, current_tools: list,
                 limit: int = 0) -> list[ToolSelectionExample]:
            """Orchestrate end-to-end mining.

            Returns full list pre-split (caller wires split + duplication via
            self.write_jsonl_outputs). For backwards-compat with simple unit
            tests, also caches the split mapping in self._last_split.

            Args:
                sessions_dir: Path to dir containing *.json session files.
                current_tools: list of ToolDescription (only .name attr used) — drives surface drift filter.
                limit: 0 = all sessions; >0 = first N.
            """
            self.metrics = self._fresh_metrics()
            current_tool_names: set[str] = {t.name for t in current_tools}
            session_paths = sorted(sessions_dir.glob("*.json"))
            if limit > 0:
                session_paths = session_paths[:limit]

            all_cands: list[Candidate] = []
            for sp in session_paths:
                try:
                    data = json.loads(sp.read_text())
                except Exception:
                    continue
                messages = data.get("messages") or []
                if not isinstance(messages, list):
                    continue
                if "error_retry" in self.signals:
                    all_cands.extend(self._extract_error_retry(messages, str(sp), current_tool_names))
                if "user_correction" in self.signals:
                    all_cands.extend(self._extract_user_correction(messages, str(sp), current_tool_names))
                if "oracle_disagreement" in self.signals and self.baseline_module is not None:
                    all_cands.extend(self._extract_oracle_disagreement(messages, str(sp), current_tool_names))

            for c in all_cands:
                self.metrics["total_candidates_by_signal"][c.signal] += 1

            # Surface drift filter (D-17)
            kept: list[Candidate] = []
            for c in all_cands:
                if c.originally_used_tool not in current_tool_names:
                    self.metrics["surface_drift_dropped"] += 1
                    self.metrics["surface_drift_tools"][c.originally_used_tool] = \
                        self.metrics["surface_drift_tools"].get(c.originally_used_tool, 0) + 1
                    continue
                kept.append(c)

            # Privacy filter (Plan 03 / D-15)
            kept2: list[Candidate] = []
            for c in kept:
                if _contains_secret(c.task) or _contains_secret(c.downstream_context):
                    self.metrics["secret_filter_skipped"] += 1
                    continue
                kept2.append(c)

            # Judge (D-03)
            verdicts: list[tuple[Candidate, Verdict]] = []
            for c in kept2:
                v = self._judge_candidate(c)
                if v.label == "confirm_misselection":
                    self.metrics["judge_confirmed_by_signal"][c.signal] += 1
                    verdicts.append((c, v))
                else:
                    self.metrics["judge_false_positives_by_signal"][c.signal] += 1

            # Reduce by hash → union signals (D-02 末尾 + Pitfall 9)
            by_hash: dict[str, ToolSelectionExample] = {}
            for c, v in verdicts:
                h = c.task_hash()
                ex = ToolSelectionExample(
                    task_description=c.task,
                    correct_tool=v.correct_tool,
                    correct_params={},
                    difficulty="medium",
                    confuser_tools=[c.originally_used_tool],
                    reason=v.rationale,
                    source="session",
                    misselection_signals=[c.signal],
                )
                if h not in by_hash:
                    by_hash[h] = ex
                else:
                    prev = by_hash[h]
                    prev.misselection_signals = sorted(set(prev.misselection_signals) | {c.signal})
                    prev.confuser_tools = sorted(set(prev.confuser_tools) | {c.originally_used_tool})

            return list(by_hash.values())

        def split_and_duplicate(
            self, examples: list[ToolSelectionExample]
        ) -> dict[str, list[ToolSelectionExample]]:
            """Bucket-split (D-13) then duplicate train-only by max multiplier (D-11).

            Returns {"train": [...], "val": [...], "holdout": [...]}.
            """
            split: dict[str, list[ToolSelectionExample]] = {"train": [], "val": [], "holdout": []}
            for ex in examples:
                h = _normalize_task_hash(ex.task_description)
                bucket = _hash_to_split(h)
                split[bucket].append(ex)
            # Train-only duplication (Pitfall 5)
            duplicated_train: list[ToolSelectionExample] = []
            for ex in split["train"]:
                n = _multiplier_for(ex.misselection_signals, self.multiplier_override)
                duplicated_train.extend([ex] * n)
            split["train"] = duplicated_train

            # Update metrics
            self.metrics["final_examples_by_split"] = {
                "train": len(set(id(e) for e in duplicated_train)),  # unique pre-dup
                "val": len(split["val"]),
                "holdout": len(split["holdout"]),
            }
            self.metrics["final_train_after_duplication"] = len(duplicated_train)
            self.metrics["multiplier_used"] = {
                k: self.multiplier_override.get(k, DEFAULT_MULTIPLIER[k])
                for k in DEFAULT_MULTIPLIER
            }
            return split

        def enumerate_candidates(
            self,
            sessions_dir: Path,
            current_tools: list,
            limit: int = 0,
        ) -> list["Candidate"]:
            """Public dry-run entry point (W4): runs the 3 extractors + surface
            drift filter + secret filter, but **skips the LLM judge**.

            Plan 05 Task 5.1 dry-run path uses this instead of poking private
            `_extract_*` methods; this isolates dry-run from internal naming.

            Args:
                sessions_dir: Path to dir containing *.json session files.
                current_tools: list of ToolDescription (only .name attr used).
                limit: 0 = all sessions; >0 = first N.

            Returns:
                List of Candidate (post drift + secret filter, pre-judge).
            """
            self.metrics = self._fresh_metrics()
            current_tool_names: set[str] = {t.name for t in current_tools}
            session_paths = sorted(sessions_dir.glob("*.json"))
            if limit > 0:
                session_paths = session_paths[:limit]
            all_cands: list[Candidate] = []
            for sp in session_paths:
                try:
                    data = json.loads(sp.read_text())
                except Exception:
                    self.metrics["jsonl_skipped_lines"] += 1
                    continue
                messages = data.get("messages") or []
                if not isinstance(messages, list):
                    continue
                if "error_retry" in self.signals:
                    all_cands.extend(self._extract_error_retry(messages, str(sp), current_tool_names))
                if "user_correction" in self.signals:
                    all_cands.extend(self._extract_user_correction(messages, str(sp), current_tool_names))
                if "oracle_disagreement" in self.signals and self.baseline_module is not None:
                    all_cands.extend(self._extract_oracle_disagreement(messages, str(sp), current_tool_names))
            for c in all_cands:
                self.metrics["total_candidates_by_signal"][c.signal] += 1
            # Surface drift filter (D-17)
            kept: list[Candidate] = []
            for c in all_cands:
                if c.originally_used_tool not in current_tool_names:
                    self.metrics["surface_drift_dropped"] += 1
                    self.metrics["surface_drift_tools"][c.originally_used_tool] = \
                        self.metrics["surface_drift_tools"].get(c.originally_used_tool, 0) + 1
                    continue
                kept.append(c)
            # Privacy filter (D-15)
            kept2: list[Candidate] = []
            for c in kept:
                if _contains_secret(c.task) or _contains_secret(c.downstream_context):
                    self.metrics["secret_filter_skipped"] += 1
                    continue
                kept2.append(c)
            return kept2
    ```
