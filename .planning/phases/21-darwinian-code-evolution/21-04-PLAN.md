<!-- D-12 size_component 公式（CONTEXT.md lines 76-87）-->
```python
# D-12 三段映射（默认 ×1.2/×1.5 阈值，executor 可调 ×1.3/×1.6）
ratio = evolved_size_bytes / baseline_size_bytes
if ratio <= 1.0:
    size_component = 1.0
elif ratio <= 1.2:  # 可放宽到 1.3
    size_component = 1.0 - (ratio - 1.0) * 1.5   # 1.0 → 0.7 线性
elif ratio <= 1.5:  # 硬上限不变
    size_component = 0.7 - (ratio - 1.2) * (0.7 / 0.3)   # 0.7 → 0.0 线性
else:
    size_component = 0.0; decision = "reject"  # >×1.5 硬 reject
```

<!-- ruff JSON 输出格式（RESEARCH.md lines 429-450，已验证）-->
```python
def _run_ruff(evolved_path: Path) -> tuple[int, list[dict]]:
    result = subprocess.run(
        ["ruff", "check", "--output-format=json", str(evolved_path)],
        capture_output=True, text=True, timeout=10
    )
    # ruff exit code 1 = 有 violations（正常），2 = 内部错误
    try:
        findings = json.loads(result.stdout) if result.stdout.strip() else []
    except json.JSONDecodeError:
        findings = []
    return len(findings), [{"rule_id": f["code"], "message": f["message"], "line": f["location"]["row"]} for f in findings]
```
